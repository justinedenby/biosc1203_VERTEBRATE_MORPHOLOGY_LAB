# ── helper functions ──────────────────────────────────────────────────────────

read_csv_flexible <- function(path) {
  tryCatch({
    df <- read_csv(path, show_col_types = FALSE)
    if (ncol(df) < 3) df <- read_delim(path, delim = ";", show_col_types = FALSE)
    df
  }, error = function(e) NULL)
}

classify_landmarks <- function(df, label_col, x_col, y_col, z_col) {
  df |>
    select(label = all_of(label_col),
           x     = all_of(x_col),
           y     = all_of(y_col),
           z     = all_of(z_col)) |>
    mutate(
      label = as.character(label),
      label = if_else(is.na(label) | label == "",
                      paste0("unlabeled_", row_number()), label),
      x = as.numeric(x),
      y = as.numeric(y),
      z = as.numeric(z),
      type = case_when(
        str_detect(label, "^OC_")        ~ "semi",
        str_detect(label, "^unlabeled_") ~ "semi",
        TRUE                              ~ "fixed"
      )
    ) |>
    filter(!is.na(x), !is.na(y), !is.na(z))
}

# TPS format:  LM3=n / x  y  z (tab-delimited, width-padded) / ID=name
build_tps <- function(lm_df, specimen_name) {
  n   <- nrow(lm_df)
  fmt <- function(v) formatC(v, format = "f", digits = 4, width = 12, flag = " ")
  coord_lines <- paste(fmt(lm_df$x), fmt(lm_df$y), fmt(lm_df$z), sep = "  ")
  c(paste0("LM3=", n), coord_lines, paste0("ID=", specimen_name))
}

# ── metadata helpers ──────────────────────────────────────────────────────────

# Load the metadata table from disk (or return empty tibble if not yet created)
load_metadata_table <- function() {
  if (file.exists(METADATA_FILE)) {
    tryCatch(
      read_csv(METADATA_FILE, show_col_types = FALSE),
      error = function(e) tibble(
        specimen = character(), habitat = character(),
        activity = character(), locomotion = character(), notes = character(),
        saved_at = character()
      )
    )
  } else {
    tibble(
      specimen   = character(),
      habitat    = character(),
      activity   = character(),
      locomotion = character(),
      notes      = character(),
      saved_at   = character()
    )
  }
}

# Upsert one row (replace existing entry for this specimen, or append)
upsert_metadata <- function(tbl, specimen, habitat, activity, locomotion, notes) {
  new_row <- tibble(
    specimen   = specimen,
    habitat    = habitat,
    activity   = activity,
    locomotion = locomotion,
    notes      = notes,
    saved_at   = format(Sys.time(), "%Y-%m-%d %H:%M:%S")
  )
  tbl <- tbl |> filter(specimen != !!specimen)
  bind_rows(tbl, new_row)
}

# ── section 4 & 5: TPS upload → GPA + PCA ────────────────────────────────────

# Parse an uploaded TPS file into a named list of coordinate matrices
# Returns list(coords_array = p×3×n array, specimen_ids = character vector,
#              lm_types = character vector of length p)
parse_tps_file <- function(path) {
  lines <- readLines(path, warn = FALSE)
  lines <- lines[nzchar(trimws(lines))]  # drop blank lines

  specimens  <- list()
  spec_ids   <- character()
  i          <- 1

  while (i <= length(lines)) {
    line <- trimws(lines[i])

    if (grepl("^LM3?=", line, ignore.case = TRUE)) {
      n_lm <- as.integer(sub("^LM3?=", "", line, ignore.case = TRUE))
      coords <- matrix(NA_real_, nrow = n_lm, ncol = 3)

      for (j in seq_len(n_lm)) {
        i <- i + 1
        vals <- as.numeric(strsplit(trimws(lines[i]), "\\s+")[[1]])
        coords[j, ] <- vals[1:3]
      }

      # Scan ahead for ID= line (skip any SCALE= lines)
      spec_id <- paste0("specimen_", length(specimens) + 1)
      while (i < length(lines)) {
        i <- i + 1
        next_line <- trimws(lines[i])
        if (grepl("^ID=", next_line, ignore.case = TRUE)) {
          spec_id <- sub("^ID=", "", next_line, ignore.case = TRUE)
          break
        }
        if (grepl("^LM3?=", next_line, ignore.case = TRUE)) {
          i <- i - 1   # back up — next iteration will catch this LM= line
          break
        }
      }

      specimens[[length(specimens) + 1]] <- coords
      spec_ids <- c(spec_ids, spec_id)

    } else {
      i <- i + 1
    }
  }

  if (length(specimens) == 0) return(NULL)

  p <- nrow(specimens[[1]])
  n <- length(specimens)

  # Build p × 3 × n array
  A <- array(NA_real_, dim = c(p, 3, n))
  for (k in seq_len(n)) A[, , k] <- specimens[[k]]
  dimnames(A)[[3]] <- spec_ids

  # Classify landmark type from first specimen's row count:
  # we don't have label info in a bare TPS, so type = "fixed" for all
  # (sliders are handled by index position passed to gpagen)
  list(A = A, ids = spec_ids, n_lm = p, n_spec = n)
}

tps_parsed <- eventReactive(input$tps_file, {

})

tps_results <- eventReactive(input$run_tps_analysis, {
  req(tps_parsed())
  tp <- tps_parsed()

  validate(
    need(tp$n_spec >= 2,
         paste0("GPA requires at least 2 specimens. Your TPS file has ",
                tp$n_spec, ". Combine more student TPS files into one and re-upload."))
  )

  # Build sliders for semilandmarks if requested.
  # Without label info in a raw TPS, we can't auto-detect semi vs fixed.
  # Sliders must be defined externally or disabled here.
  sliders <- NULL
  # (sliders will be wired up once A provides the row indices)

  gpa <- tryCatch(
    gpagen(tp$A, curves = sliders,
           ProcD = (input$tps_procD == "full"),
           print.progress = FALSE),
    error = function(e) {
      validate(need(FALSE, paste("GPA failed:", e$message)))
    }
  )

  pca <- tryCatch(
    gm.prcomp(gpa$coords),
    error = function(e) {
      validate(need(FALSE, paste("PCA failed:", e$message)))
    }
  )

  list(gpa = gpa, pca = pca, ids = tp$ids)
})
