# ── Server ────────────────────────────────────────────────────────────────────
server <- function(input, output, session) {

  # Install packages in WebR on first run (runtime, browser-side).
  # DT and plotly are intentionally NOT used — their htmlwidgets runtime
  # throws a "data is null / deps" error under shinylive. Tables are rendered
  # as plain HTML and plots as static base/ggplot images instead.
  for (pkg in c("geomorph","ggplot2","readr","dplyr","stringr","ape","nlme")) {
    if (!requireNamespace(pkg, quietly = TRUE)) webr::install(pkg)
  }

  library(geomorph); library(ggplot2)
  library(readr);    library(dplyr); library(stringr)
  library(ape);      library(nlme)

  # ── helpers ─────────────────────────────────────────────────────────────────

  # Render a data frame as a simple scrollable HTML table (replaces DT).
  html_table <- function(df, digits = 4) {
    df <- as.data.frame(df)
    num <- vapply(df, is.numeric, logical(1))
    df[num] <- lapply(df[num], function(v) round(v, digits))
    hdr <- paste0("<th>", names(df), "</th>", collapse = "")
    rows <- apply(df, 1, function(r)
      paste0("<tr>", paste0("<td>", r, "</td>", collapse = ""), "</tr>"))
    HTML(paste0(
      "<div class='app-table'><table><thead><tr>", hdr,
      "</tr></thead><tbody>", paste0(rows, collapse = ""),
      "</tbody></table></div>"
    ))
  }

  read_csv_flexible <- function(path) {
    tryCatch({
      df <- read_csv(path, show_col_types = FALSE)
      if (ncol(df) < 3) df <- read_delim(path, delim = ";",
                                          show_col_types = FALSE)
      df
    }, error = function(e) NULL)
  }

  norm_species <- function(x) {
    x <- trimws(as.character(x))
    x <- gsub("\\s+", "_", x)
    tolower(x)
  }

  classify_landmarks <- function(df, label_col, x_col, y_col, z_col) {
    df |>
      select(label = all_of(label_col), x = all_of(x_col),
             y = all_of(y_col),         z = all_of(z_col)) |>
      mutate(
        label = as.character(label),
        label = if_else(is.na(label) | label == "",
                        paste0("unlabeled_", row_number()), label),
        x = as.numeric(x), y = as.numeric(y), z = as.numeric(z),
        type = case_when(
          str_detect(label, "^OC_")        ~ "semi",
          str_detect(label, "^unlabeled_") ~ "semi",
          TRUE                              ~ "fixed"
        )
      ) |>
      filter(!is.na(x), !is.na(y), !is.na(z))
  }

  build_tps <- function(lm_df, specimen_name) {
    fmt <- function(v) formatC(v, format = "f", digits = 4, width = 12,
                                flag = " ")
    coord_lines <- paste(fmt(lm_df$x), fmt(lm_df$y), fmt(lm_df$z), sep = "  ")
    c(paste0("LM3=", nrow(lm_df)), coord_lines, paste0("ID=", specimen_name))
  }

  parse_tps_file <- function(path) {
    lines <- readLines(path, warn = FALSE)
    lines <- lines[nzchar(trimws(lines))]
    specimens <- list(); spec_ids <- character(); i <- 1
    while (i <= length(lines)) {
      line <- trimws(lines[i])
      if (grepl("^LM3?=", line, ignore.case = TRUE)) {
        n_lm   <- as.integer(sub("^LM3?=", "", line, ignore.case = TRUE))
        coords <- matrix(NA_real_, nrow = n_lm, ncol = 3)
        for (j in seq_len(n_lm)) {
          i <- i + 1
          vals <- as.numeric(strsplit(trimws(lines[i]), "\\s+")[[1]])
          coords[j, ] <- vals[1:3]
        }
        spec_id <- paste0("specimen_", length(specimens) + 1)
        while (i < length(lines)) {
          i <- i + 1; next_line <- trimws(lines[i])
          if (grepl("^ID=", next_line, ignore.case = TRUE)) {
            spec_id <- sub("^ID=", "", next_line, ignore.case = TRUE); break
          }
          if (grepl("^LM3?=", next_line, ignore.case = TRUE)) {
            i <- i - 1; break
          }
        }
        specimens[[length(specimens) + 1]] <- coords
        spec_ids <- c(spec_ids, spec_id)
      } else { i <- i + 1 }
    }
    if (length(specimens) == 0) return(NULL)
    p <- nrow(specimens[[1]]); n <- length(specimens)
    A <- array(NA_real_, dim = c(p, 3, n))
    for (k in seq_len(n)) A[, , k] <- specimens[[k]]
    dimnames(A)[[3]] <- spec_ids
    list(A = A, ids = spec_ids, n_lm = p, n_spec = n)
  }

  # The tree now comes from the user's upload (Newick, or NEXUS as a fallback)
  # rather than a file bundled with the app.
  master_tree <- reactive({
    validate(need(!is.null(input$tree_file),
      "Upload a tree file (Newick or NEXUS) in the sidebar to continue."))
    path <- input$tree_file$datapath
    tr <- tryCatch(suppressWarnings(read.tree(path)), error = function(e) NULL)
    if (is.null(tr)) tr <- tryCatch(suppressWarnings(read.nexus(path)),
                                    error = function(e) NULL)
    validate(
      need(!is.null(tr),
        "The tree file could not be read. Upload a Newick (.nwk, .tre, .tree) or NEXUS (.nex) file."),
      need(!inherits(tr, "multiPhylo"),
        "The file contains more than one tree. Upload a single tree."),
      need(!is.null(tr$edge.length),
        "The tree has no branch lengths, which the phylogenetic models require."),
      need(all(tr$edge.length >= 0),
        "The tree has negative branch lengths."),
      need(length(tr$tip.label) >= 3,
        "The tree needs at least 3 tips."))
    tr
  })

  # Cache the normalized tip labels once so every downstream output
  # (match summary, both tree plots, PGLS) reuses them instead of
  # recomputing norm_species() over the whole tree each render — this
  # matters under WebR where each call is slow.
  master_tips_norm <- reactive({
    tr <- master_tree(); req(tr)
    norm_species(tr$tip.label)
  })

  # ── section 1 & 2: CSV / parse / TPS ────────────────────────────────────────

  raw_data <- reactive({ req(input$csv_file); read_csv_flexible(input$csv_file$datapath) })

  output$col_pickers <- renderUI({
    df <- raw_data(); req(df); cols <- names(df)
    guess <- function(p) {
      m <- cols[str_detect(tolower(cols), paste(tolower(p), collapse = "|"))]
      if (length(m)) m[1] else cols[1]
    }
    tagList(
      selectInput("col_label", "Label column", choices = cols,
                  selected = guess(c("label","name","id","lm"))),
      selectInput("col_x", "X / R coord", choices = cols,
                  selected = guess(c("^r$","_x$","x$","r$"))),
      selectInput("col_y", "Y / A coord", choices = cols,
                  selected = guess(c("^a$","_y$","y$","a$"))),
      selectInput("col_z", "Z / S coord", choices = cols,
                  selected = guess(c("^s$","_z$","z$","s$")))
    )
  })

  output$tbl_raw <- renderUI({ req(raw_data()); html_table(head(raw_data(), 100)) })
  output$tbl_raw_ready <- reactive({
    isTRUE(tryCatch(!is.null(raw_data()), error = function(e) FALSE))
  })
  outputOptions(output, "tbl_raw_ready", suspendWhenHidden = FALSE)

  parsed <- eventReactive(input$run_parse, {
    req(raw_data(), input$col_label, input$col_x, input$col_y, input$col_z)
    classify_landmarks(raw_data(), input$col_label,
                       input$col_x, input$col_y, input$col_z)
  })

  output$tbl_parsed <- renderUI({ req(parsed()); html_table(parsed()) })

  output$lm_summary <- renderPrint({
    req(parsed()); df <- parsed()
    cat("Total landmarks:", nrow(df), "\n")
    cat("Fixed landmarks:", sum(df$type == "fixed"), "\n")
    cat("Semilandmarks  :", sum(df$type == "semi"),  "\n\n")
    cat("Fixed landmark labels:\n")
    cat(paste(" ", df$label[df$type == "fixed"], collapse = "\n"), "\n\n")
    cat("Semilandmark labels:\n")
    cat(paste(" ", df$label[df$type == "semi"],  collapse = "\n"), "\n")
  })

  tps_lines <- reactive({ req(parsed()); build_tps(parsed(), input$specimen_name) })
  output$tps_preview <- renderText({ req(tps_lines()); paste(tps_lines(), collapse = "\n") })
  output$dl_tps <- downloadHandler(
    filename = function() paste0(input$specimen_name, ".tps"),
    content  = function(file) writeLines(tps_lines(), file)
  )

  # ── section 3: TPS analysis ─────────────────────────────────────────────────

  tps_parsed <- eventReactive(input$tps_file, {
    req(input$tps_file)
    result <- tryCatch(parse_tps_file(input$tps_file$datapath),
                       error = function(e) NULL)
    validate(need(!is.null(result),
      "Could not parse TPS file. Check format: LM3=n / coords / ID=name."))
    result
  })

  output$tbl_tps_specimens <- renderUI({
    req(tps_parsed()); tp <- tps_parsed()
    html_table(data.frame(Index = seq_along(tp$ids), Specimen = tp$ids,
                          N_landmarks = tp$n_lm))
  })
  output$tbl_tps_ready <- reactive({
    isTRUE(tryCatch(!is.null(input$tps_file), error = function(e) FALSE))
  })
  outputOptions(output, "tbl_tps_ready", suspendWhenHidden = FALSE)

  tps_results <- eventReactive(input$run_tps_analysis, {
    req(tps_parsed()); tp <- tps_parsed()
    validate(need(tp$n_spec >= 2,
      paste0("GPA requires at least 2 specimens. Your file has ", tp$n_spec, ".")))
    gpa <- tryCatch(
      gpagen(tp$A, curves = NULL, ProcD = (input$tps_procD == "full"),
             print.progress = FALSE),
      error = function(e) validate(need(FALSE, paste("GPA failed:", e$message))))
    pca <- tryCatch(gm.prcomp(gpa$coords),
      error = function(e) validate(need(FALSE, paste("PCA failed:", e$message))))
    list(gpa = gpa, pca = pca, ids = tp$ids)
  })

  output$tps_gpa_summary <- renderPrint({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); res <- tps_results()
    cat("=== GPA (Generalized Procrustes Analysis) ===\n\n")
    print(summary(res$gpa))
    cat("\nConsensus shape (first 6 landmarks):\n")
    print(round(head(res$gpa$consensus, 6), 4))
  })

  output$tbl_tps_pca <- renderUI({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); pca <- tps_results()$pca
    eigs <- pca$sdev^2; var_exp <- round(eigs / sum(eigs) * 100, 2)
    n_pcs <- min(input$tps_n_pcs, length(var_exp))
    scores_df <- as.data.frame(pca$x[, 1:n_pcs, drop = FALSE])
    names(scores_df) <- paste0("PC", 1:n_pcs, " (", var_exp[1:n_pcs], "%)")
    scores_df <- cbind(Specimen = rownames(scores_df), scores_df)
    html_table(scores_df)
  })

  output$plot_tps_pca <- renderPlot({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); pca <- tps_results()$pca
    eigs <- pca$sdev^2; var_exp <- round(eigs / sum(eigs) * 100, 2)
    scores <- as.data.frame(pca$x); scores$Specimen <- rownames(scores)
    ggplot(scores, aes(x = Comp1, y = Comp2, label = Specimen)) +
      geom_point(size = 3, colour = "#2c7bb6") +
      geom_text(vjust = -0.8, size = 3) +
      labs(x = paste0("PC1 (", var_exp[1], "% variance)"),
           y = paste0("PC2 (", var_exp[2], "% variance)"),
           title = "PCA morphospace — PC1 vs PC2") +
      theme_minimal(base_size = 13)
  })

  output$plot_tps_scree <- renderPlot({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); pca <- tps_results()$pca
    eigs <- pca$sdev^2; var_exp <- round(eigs / sum(eigs) * 100, 2)
    n_show <- min(input$tps_n_pcs, length(var_exp))
    df_sc <- data.frame(
      PC = factor(paste0("PC", 1:n_show), levels = paste0("PC", 1:n_show)),
      Variance = var_exp[1:n_show])
    ggplot(df_sc, aes(x = PC, y = Variance)) +
      geom_col(fill = "#2c7bb6", alpha = 0.8) +
      geom_line(aes(group = 1), colour = "#d7191c", linewidth = 0.8) +
      geom_point(colour = "#d7191c", size = 2) +
      labs(title = "Scree plot", y = "% Variance", x = "") +
      theme_minimal(base_size = 13)
  })

  output$plot_tps_consensus <- renderPlot({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); cons <- tps_results()$gpa$consensus
    df_plot <- data.frame(x = cons[,1], y = cons[,2], z = cons[,3],
                          label = rownames(cons))
    par(mfrow = c(1, 2), mar = c(4, 4, 3, 1))
    plot(df_plot$x, df_plot$y, pch = 19, col = "#2c7bb6", cex = 1.2,
         xlab = "X (R)", ylab = "Y (A)", main = "Consensus — R-A view")
    text(df_plot$x, df_plot$y, labels = df_plot$label, cex = 0.6, pos = 3)
    plot(df_plot$x, df_plot$z, pch = 19, col = "#2c7bb6", cex = 1.2,
         xlab = "X (R)", ylab = "Z (S)", main = "Consensus — R-S view")
    text(df_plot$x, df_plot$z, labels = df_plot$label, cex = 0.6, pos = 3)
  })

  # ── section 4: phylogenetic analyses (PGLS) ─────────────────────────────────

  trait_data <- reactive({
    req(input$trait_file)
    df <- read_csv_flexible(input$trait_file$datapath)
    validate(need(!is.null(df) && ncol(df) >= 2,
      "Trait CSV needs a species column plus at least one trait column."))
    names(df)[1] <- "species_raw"
    df$species_match <- norm_species(df$species_raw)
    df
  })

  output$trait_ready <- reactive({
    isTRUE(tryCatch(!is.null(input$trait_file), error = function(e) FALSE))
  })
  outputOptions(output, "trait_ready", suspendWhenHidden = FALSE)

  trait_cols <- reactive({
    df <- trait_data()
    setdiff(names(df), c("species_raw", "species_match"))
  })

  # ── Trait classification ────────────────────────────────────────────────────
  # Every trait column is classified as one of:
  #   "numeric"     numeric with 3+ distinct values (continuous)
  #   "binary"      exactly 2 distinct non-missing values: numeric 0/1 codes,
  #                 TRUE/FALSE, or two-level text such as present/absent
  #   "categorical" anything else (e.g. text with 3+ levels)
  trait_kind <- function(v) {
    n_distinct <- length(unique(v[!is.na(v)]))
    if (n_distinct == 2) {
      "binary"
    } else if (is.numeric(v)) {
      "numeric"
    } else {
      "categorical"
    }
  }

  trait_kinds <- reactive({
    df <- trait_data(); cols <- trait_cols()
    vapply(cols, function(cc) trait_kind(df[[cc]]), character(1))
  })

  # Collapsed view used for predictors: binary traits are handled as
  # categorical (group comparison), everything else keeps its type.
  trait_types <- reactive({
    k <- trait_kinds()
    setNames(ifelse(k == "numeric", "numeric", "categorical"), names(k))
  })

  # The two values of a binary trait, in sorted order (e.g. "0" "1").
  binary_levels <- function(v) as.character(sort(unique(v[!is.na(v)])))

  # Names of the binary traits (used for labelling).
  binary_traits <- reactive({
    k <- trait_kinds(); names(k)[k == "binary"]
  })

  output$tbl_traits <- renderUI({
    req(trait_data())
    html_table(trait_data() |> select(-species_match))
  })

  # Response may be continuous (PGLS) or binary (logistic / Pagel).
  output$pgls_response_ui <- renderUI({
    req(trait_data())
    kinds    <- trait_kinds()
    num_cols <- names(kinds)[kinds == "numeric"]
    bin_cols <- names(kinds)[kinds == "binary"]
    validate(need(length(num_cols) + length(bin_cols) >= 1,
      "Need at least one numeric or binary (presence/absence) trait to use as the response."))

    grouped <- list()
    if (length(num_cols))
      grouped[["Continuous → PGLS"]] <- as.list(num_cols)
    if (length(bin_cols))
      grouped[["Binary (presence/absence) → logistic / Pagel"]] <- as.list(bin_cols)

    selectInput("pgls_response", "Response (Y)",
                choices = grouped, selected = c(num_cols, bin_cols)[1])
  })

  # For a binary response: which of the two values is the "event" (coded 1).
  output$pgls_event_ui <- renderUI({
    req(trait_data(), input$pgls_response)
    kinds <- trait_kinds()
    idx   <- match(input$pgls_response, names(kinds))
    if (is.na(idx) || kinds[[idx]] != "binary") return(NULL)
    lv <- binary_levels(trait_data()[[input$pgls_response]])
    selectInput("pgls_event", "Outcome modeled as 1 (presence / event)",
                choices = lv, selected = lv[length(lv)])
  })

  output$pgls_predictor_ui <- renderUI({
    req(trait_data(), input$pgls_response)
    tt      <- trait_types()
    choices <- setdiff(trait_cols(), input$pgls_response)
    validate(need(length(choices) >= 1, "Need at least one other trait as predictor."))

    num_choices <- choices[tt[choices] == "numeric"]
    cat_choices <- choices[tt[choices] == "categorical"]

    # Group the dropdown so students see the type of each predictor.
    # Numeric predictors are listed first and selected by default so the
    # regression line shows unless they deliberately pick a category.
    grouped <- list()

  # Install a package on first use (browser-side). Used for the packages that

  ensure_pkg <- function(pkg) {
    if (requireNamespace(pkg, quietly = TRUE)) return(TRUE)
    ok <- tryCatch({ webr::install(pkg); TRUE }, error = function(e) FALSE)
    ok && requireNamespace(pkg, quietly = TRUE)
  }

  # Evaluate an expression, collecting warnings instead of printing them.
  with_warnings <- function(expr) {
    w <- character()
    val <- withCallingHandlers(expr,
      warning = function(cond) {
        w <<- c(w, conditionMessage(cond))
        invokeRestart("muffleWarning")
      })
    list(value = val, warnings = unique(w))
  }

  # Shared header lines for the two summary tabs.
  analysis_header <- function(res) {

             res$nonevent, "' = 0)")

  # Caution shown when a binary-response dataset is small or lopsided.
  binary_caution <- function(res) {
    n <- nrow(res$mf); n_min <- min(sum(res$mf$y == 1), sum(res$mf$y == 0))
    if (n < 10 || n_min < 5) {
      cat("\nCAUTION: binary outcomes carry little information per species. With\n")
      cat("fewer than 10 species, or fewer than 5 in the rarer outcome, estimates\n")
      cat("and p-values are unstable. Interpret with great caution.\n")
    }
  }

  # Pagel's (1994) test of correlated evolution for two binary traits,
  # implemented with ape::ace so no extra package is needed.
  #   Combined states: 1 = x1|y1, 2 = x1|y2, 3 = x2|y1, 4 = x2|y2
  #   Independent model: 4 rates (x and y each evolve at rates that do not
  #                      depend on the state of the other trait)
  #   Dependent model  : 8 rates (every single-trait change has its own rate)
  #   Simultaneous changes in both traits are not allowed in either model.
  pagel_test <- function(tree, x, y) {
    x <- factor(x); y <- factor(y)
    stopifnot(nlevels(x) == 2, nlevels(y) == 2)
    lx <- levels(x); ly <- levels(y)
    lab <- c(paste(lx[1], ly[1], sep = "|"), paste(lx[1], ly[2], sep = "|"),
             paste(lx[2], ly[1], sep = "|"), paste(lx[2], ly[2], sep = "|"))
    st <- factor(paste(as.character(x), as.character(y), sep = "|"),
                 levels = lab)

    dep <- matrix(0L, 4, 4)
    dep[1, 2] <- 1L; dep[1, 3] <- 2L
    dep[2, 1] <- 3L; dep[2, 4] <- 4L
    dep[3, 1] <- 5L; dep[3, 4] <- 6L
    dep[4, 2] <- 7L; dep[4, 3] <- 8L

    ind <- matrix(0L, 4, 4)
    ind[1, 2] <- 1L; ind[3, 4] <- 1L   # y: first -> second state
    ind[2, 1] <- 2L; ind[4, 3] <- 2L   # y: second -> first state
    ind[1, 3] <- 3L; ind[2, 4] <- 3L   # x: first -> second state
    ind[3, 1] <- 4L; ind[4, 2] <- 4L   # x: second -> first state

    fit_dep <- ape::ace(unname(st), tree, type = "discrete", model = dep)
    fit_ind <- ape::ace(unname(st), tree, type = "discrete", model = ind)

    make_Q <- function(fit, M) {
      Q <- matrix(0, 4, 4, dimnames = list(lab, lab))
      for (i in 1:4) for (j in 1:4)
        if (M[i, j] > 0) Q[i, j] <- fit$rates[M[i, j]]
      diag(Q) <- -rowSums(Q)
      Q
    }

    LR_raw <- 2 * (fit_dep$loglik - fit_ind$loglik)
    df     <- max(dep) - max(ind)
    LR     <- max(LR_raw, 0)
    list(labels = lab, logL_ind = fit_ind$loglik, logL_dep = fit_dep$loglik,
         AIC_ind = -2 * fit_ind$loglik + 2 * max(ind),
         AIC_dep = -2 * fit_dep$loglik + 2 * max(dep),
         Q_ind = make_Q(fit_ind, ind), Q_dep = make_Q(fit_dep, dep),
         LR = LR, LR_raw = LR_raw, df = df,
         P = pchisq(LR, df, lower.tail = FALSE))
  }

  pgls_fit <- eventReactive(input$run_pgls, {
    req(trait_data(), pruned_tree(), input$pgls_response, input$pgls_predictor)
    tr <- pruned_tree(); df <- trait_data()
    df <- df[df$species_match %in% norm_species(tr$tip.label), , drop = FALSE]
    df <- df[!duplicated(df$species_match), , drop = FALSE]
    rownames(df) <- df$species_match
    tr$tip.label <- norm_species(tr$tip.label)

    resp <- input$pgls_response; pred <- input$pgls_predictor
    kinds     <- trait_kinds()
    resp_kind <- kinds[[resp]]
    pred_kind <- kinds[[pred]]
    pred_type <- trait_types()[[pred]]

    # Binary response: 1 = the chosen "event" (presence), 0 = the other value.
    event <- NULL; nonevent <- NULL
    if (resp_kind == "binary") {
      lv    <- binary_levels(trait_data()[[resp]])
      event <- input$pgls_event
      validate(need(!is.null(event) && event %in% lv,
        "Choose which outcome of the binary response is coded as 1 (presence)."))
      nonevent <- setdiff(lv, event)
    }

    yraw <- df[[resp]]
    y <- if (resp_kind == "binary") {
      ifelse(is.na(yraw), NA_integer_, as.integer(as.character(yraw) == event))
    } else {
      as.numeric(yraw)
    }

    mf <- data.frame(
      species = rownames(df),
      y = y,
      x = if (pred_type == "numeric") as.numeric(df[[pred]])
          else factor(df[[pred]]),
      stringsAsFactors = FALSE
    )
    rownames(mf) <- mf$species
    mf <- mf[complete.cases(mf[, c("y", "x")]), , drop = FALSE]
    if (is.factor(mf$x)) mf$x <- droplevels(mf$x)

    # Hard floor: PGLS needs to estimate an intercept, a slope, and a
    # residual variance, so it cannot run below 4 species and is not
    # trustworthy much below that. Block clearly rather than fit garbage.
    validate(need(nrow(mf) >= 4,
      paste0("Only ", nrow(mf), " species have complete data for this pair. ",
             "PGLS needs at least 4 (and ideally 10+) to give a meaningful fit. ",
             "Add more species or choose traits with fewer missing values.")))

    # Binary response: both outcomes must be represented, and binary data
    # need more species than continuous data to say anything.
    if (resp_kind == "binary") {
      n_event <- sum(mf$y == 1); n_non <- sum(mf$y == 0)
      validate(need(nrow(mf) >= 6 && n_event >= 2 && n_non >= 2,
        paste0("A binary response needs both outcomes represented by at least ",
               "2 species (and at least 6 species overall). Current counts for '",
               resp, "': ", event, " = ", n_event, ", ", nonevent, " = ", n_non,
               ". Choose a different response or add species.")))
      mf$y_lab <- factor(ifelse(mf$y == 1, event, nonevent),
                         levels = c(nonevent, event))
    }

    # For a categorical predictor, every group needs at least 2 species,
    # otherwise gls cannot estimate that group and the fit fails or misleads.
    if (trait_types()[[pred]] == "categorical") {
      tab <- table(mf$x)
      validate(need(all(tab >= 2) && length(tab) >= 2,
        paste0("Each category needs at least 2 species (and there must be at ",
               "least 2 categories). Current counts: ",
               paste(names(tab), as.integer(tab), sep = "=", collapse = ", "),
               ". Pick a different predictor or add species.")))
    }

    tr2 <- drop.tip(tr, setdiff(tr$tip.label, rownames(mf)))
    mf  <- mf[tr2$tip.label, , drop = FALSE]

    # Which analysis applies to this response / predictor combination?
    analysis <- if (resp_kind != "binary") {
      "continuous"                       # PGLS (gls)
    } else if (pred_type == "numeric") {
      "bin_num"                          # logistic regression / phyloglm
    } else if (pred_kind == "binary") {
      "bin_bin"                          # 2x2 table / Pagel's test
    } else {
      "bin_cat"                          # r x 2 table / phyloglm
    }

    fit_warnings <- character()
    err_obj <- function(e) structure(list(msg = conditionMessage(e)),
                                     class = "pgls_error")

    if (analysis == "continuous") {
      cor_struct <- if (input$pgls_model == "lambda") {
        corPagel(value = 0.5, phy = tr2, form = ~species, fixed = FALSE)
      } else {
        corBrownian(phy = tr2, form = ~species)
      }

      # Fit PGLS. If corPagel fails to optimise lambda (common on small or
      # low-signal data), retry with lambda fixed at a middling value so the
      # student still gets a result instead of a blank screen. Capture any
      # error message so it can be shown rather than silently swallowed.
      fit <- tryCatch(
        gls(y ~ x, data = mf, correlation = cor_struct, method = "ML"),
        error = function(e) {
          tryCatch(
            gls(y ~ x, data = mf,
                correlation = corPagel(0.5, phy = tr2, form = ~species,
                                       fixed = TRUE),
                method = "ML"),
            error = function(e2)
              structure(list(msg = conditionMessage(e)), class = "pgls_error")
          )
        })
    } else if (analysis %in% c("bin_num", "bin_cat")) {
      # Phylogenetic logistic regression (phylolm::phyloglm)
      fw <- with_warnings(tryCatch(
        withProgress(message = "Fitting phylogenetic logistic regression",
                     detail = "(the first use also installs phylolm)",
                     value = 0.5, {
          if (!ensure_pkg("phylolm"))
            stop("The 'phylolm' package could not be loaded in this browser session.")
          phylolm::phyloglm(y ~ x, data = mf, phy = tr2,
                            method = "logistic_MPLE")
        }),
        error = err_obj))
      fit <- fw$value; fit_warnings <- fw$warnings
    } else {
      # Pagel's (1994) test of correlated evolution (ape::ace; see pagel_test)
      fw <- with_warnings(tryCatch(
        withProgress(message = "Fitting Pagel's test of correlated evolution",
                     detail = "(fits two 4-state models; may take a few seconds)",
                     value = 0.5,
                     pagel_test(tr2, x = mf$x, y = mf$y_lab)),
        error = err_obj))
      fit <- fw$value; fit_warnings <- fw$warnings
    }

    list(fit = fit, mf = mf, resp = resp, pred = pred,
         pred_type = pred_type, pred_kind = pred_kind, resp_kind = resp_kind,
         analysis = analysis, event = event, nonevent = nonevent,
         fit_warnings = fit_warnings)
  })

  # Phylogeny-naive comparison analysis. Uses exactly the same species/data
  # (res$mf) as the phylogenetic fit so the two summaries are comparable.
  #   continuous response, numeric predictor     -> OLS            (lm)
  #   continuous response, categorical predictor -> one-way ANOVA  (aov)
  #   binary response, numeric predictor         -> logistic regression (glm)
  #   binary response, categorical/binary pred.  -> contingency table + Fisher's exact test
  output$ols_summary <- renderPrint({
    validate(need(input$run_pgls > 0, "Choose variables and click 'Run analysis'."))
    req(pgls_fit()); res <- pgls_fit(); mf <- res$mf

    if (res$analysis == "continuous") {
      if (res$pred_type == "numeric") {
        cat("=== Ordinary Least Squares (phylogeny-naive; stats::lm) ===\n")
        analysis_header(res)
        cat("Model      : y ~ x   (y = response, x = predictor)\n\n")
        print(summary(lm(y ~ x, data = mf)))
      } else {
        cat("=== One-way ANOVA (phylogeny-naive; stats::aov) ===\n")
        analysis_header(res)
        cat("Model      : y ~ x   (y = response, x = predictor)\n\n")
        print(summary(aov(y ~ x, data = mf)))
      }

    } else if (res$analysis == "bin_num") {
      cat("=== Logistic Regression (phylogeny-naive; stats::glm, binomial) ===\n")
      analysis_header(res)
      cat("Model      : y ~ x   (y = 1 for '", res$event, "', x = predictor)\n\n", sep = "")
      gw <- with_warnings(glm(y ~ x, data = mf, family = binomial))
      print(summary(gw$value))
      cat("\nOdds ratios with 95% Wald confidence intervals:\n")
      tryCatch(print(round(exp(cbind(OddsRatio = coef(gw$value),
                                     confint.default(gw$value))), 3)),
               error = function(e) cat("  (could not be computed)\n"))
      cat("\nAn odds ratio above 1 means higher predictor values go with greater\n")
      cat("odds of '", res$event, "'; below 1 means lower odds.\n", sep = "")
      if (length(gw$warnings)) {
        cat("\nWARNINGS from glm():\n")
        for (w in gw$warnings) cat("  -", w, "\n")
        if (any(grepl("0 or 1", gw$warnings)))
          cat("\nThis usually means the predictor (almost) perfectly separates the two\noutcomes ('separation'). The coefficients and p-values are then\nunreliable; rely on the plot and be cautious.\n")
      }
      binary_caution(res)

    } else {
      cat("=== Contingency table + Fisher's exact test (phylogeny-naive) ===\n")
      analysis_header(res)
      cat("\n")
      tab <- table(mf$x, mf$y_lab, dnn = c(res$pred, res$resp))
      cat("Counts of species (rows = predictor, columns = response):\n")
      print(tab)
      cat("\nRow proportions:\n")
      print(round(prop.table(tab, 1), 3))
      ex <- tryCatch(suppressWarnings(chisq.test(tab)$expected),
                     error = function(e) NULL)
      if (!is.null(ex)) {
        cat("\nExpected counts if the two traits were independent:\n")
        print(round(ex, 2))
        if (any(ex < 5))
          cat("Some expected counts are below 5, so Fisher's exact test (below)\nis preferred over a chi-square test.\n")
      }
      ft <- tryCatch(fisher.test(tab), error = function(e)
        tryCatch(fisher.test(tab, simulate.p.value = TRUE, B = 1e5),
                 error = function(e2) NULL))
      cat("\n")
      if (is.null(ft)) cat("Fisher's exact test could not be computed.\n") else print(ft)
      binary_caution(res)
    }
    cat("\nNOTE: This analysis treats species as independent data points and\n")
    cat("ignores phylogenetic relatedness. Compare with the PGLS Summary tab.\n")
  })

  output$pgls_summary <- renderPrint({
    validate(need(input$run_pgls > 0, "Choose variables and click 'Run analysis'."))
    req(pgls_fit()); res <- pgls_fit()

    if (res$analysis == "continuous") {
      if (inherits(res$fit, "pgls_error")) {
        cat("PGLS could not be fit for this combination.\n\n")
        cat("Error reported by gls():\n  ", res$fit$msg, "\n\n")
        cat("Tips: try the 'Brownian motion' correlation instead of Pagel's\n")
        cat("lambda, or choose a different response/predictor pair.\n")
        return(invisible())
      }
      cat("=== Phylogenetic Generalized Least Squares (nlme::gls) ===\n")
      cat("Response   :", res$resp, "\n")
      cat("Predictor  :", res$pred, " (", res$pred_type, ")\n")
      cat("Correlation:", input$pgls_model, "\n")
      cat("N species  :", nrow(res$mf), "\n\n")
      if (nrow(res$mf) < 10) {
        cat("NOTE: With fewer than 10 species, PGLS estimates (especially\n")
        cat("Pagel's lambda and p-values) are unstable. Interpret with caution.\n\n")
      }
      print(summary(res$fit))

    } else if (res$analysis %in% c("bin_num", "bin_cat")) {
      if (inherits(res$fit, "pgls_error")) {
        cat("Phylogenetic logistic regression could not be fit for this combination.\n\n")
        cat("Error reported:\n  ", res$fit$msg, "\n\n")
        cat("Tips: binary data carry little information. Try a different predictor,\n")
        cat("make sure both outcomes are well represented, or add more species.\n")
        return(invisible())
      }
      cat("=== Phylogenetic Logistic Regression (phylolm::phyloglm) ===\n")
      analysis_header(res)
      cat("Method     : logistic_MPLE (phyloglm default)\n\n")
      print(summary(res$fit))
      cat("\nalpha describes how quickly phylogenetic correlation decays along the\n")
      cat("tree: a small alpha means strong phylogenetic signal in the outcome, a\n")
      cat("very large alpha means little signal (results approach ordinary\n")
      cat("logistic regression).\n")
      if (res$analysis == "bin_cat")
        cat("\nFor a multi-level predictor, each coefficient compares that level\nwith the first (reference) level; phyloglm gives no single overall test.\n")
      if (length(res$fit_warnings)) {
        cat("\nWARNINGS from phyloglm():\n")
        for (w in res$fit_warnings) cat("  -", w, "\n")
      }
      binary_caution(res)

    } else {
      if (inherits(res$fit, "pgls_error")) {
        cat("Pagel's test could not be fit for this combination.\n\n")
        cat("Error reported:\n  ", res$fit$msg, "\n\n")
        cat("Tips: Pagel's test needs enough changes between states across the tree.\n")
        cat("With few species or rare states the optimisation can fail.\n")
        return(invisible())
      }
      cat("=== Pagel (1994) test of correlated evolution (ape::ace) ===\n")
      analysis_header(res)
      f <- res$fit
      cat("\nCombined states are written predictor|response:\n  ",
          paste(f$labels, collapse = "    "), "\n\n")
      cat("Independent model (4 rates): log-likelihood =",
          round(f$logL_ind, 3), "  AIC =", round(f$AIC_ind, 3), "\n")
      cat("Dependent model   (8 rates): log-likelihood =",
          round(f$logL_dep, 3), "  AIC =", round(f$AIC_dep, 3), "\n")
      cat("\nTransition rates, independent model (rows = from, columns = to):\n")
      print(round(f$Q_ind, 4))
      cat("\nTransition rates, dependent model (rows = from, columns = to):\n")
      print(round(f$Q_dep, 4))
      cat("\nLikelihood-ratio test, dependent vs. independent evolution:\n")
      cat("  LR =", round(f$LR, 3), " on", f$df, "df,   p =", signif(f$P, 3), "\n")
      if (f$LR_raw < -1e-6) {
        cat("\nWARNING: the dependent model (which contains the independent model)\n")
        cat("came out with a LOWER likelihood, so the optimiser did not find the\n")
        cat("best fit. Treat this result as unreliable.\n")
      }
      cat("\nHow to read this: the test compares a model where the two traits evolve\n")
      cat("independently with one where the rate of change in each trait depends on\n")
      cat("the state of the other. A small p-value means the dependent model fits\n")
      cat("significantly better, i.e. the traits show correlated evolution. This is\n")
      cat("a different question from the contingency-table test on the OLS Summary\n")
      cat("tab, which only asks whether the two traits are associated across species.\n")
      if (length(res$fit_warnings)) {
        cat("\nWARNINGS:\n")
        for (w in res$fit_warnings) cat("  -", w, "\n")
      }
      binary_caution(res)
    }
  })

  output$plot_pgls <- renderPlot({
    validate(need(input$run_pgls > 0, "Choose variables and click 'Run analysis'."))
    req(pgls_fit()); res <- pgls_fit(); mf <- res$mf
    mf$species <- rownames(mf)
    failed <- inherits(res$fit, "pgls_error")

    if (res$analysis == "continuous") {
      validate(need(!failed,
        paste0("PGLS could not be fit: ", if (failed) res$fit$msg else "",
               "\nSee the PGLS summary tab for details.")))
      if (res$pred_type == "numeric") {
        # Guard: a numeric predictor needs enough distinct values for a
        # meaningful regression line. If not, tell the student rather than
        # drawing a misleading fit.
        validate(need(length(unique(mf$x)) >= 3,
          paste0("'", res$pred, "' has fewer than 3 distinct numeric values, ",
                 "so a regression line would be misleading. Pick a predictor ",
                 "with more variation, or treat this variable as categorical.")))

        # PGLS (phylogeny-corrected) coefficients
        b_pgls <- coef(res$fit)
        # Ordinary least squares for comparison (ignores phylogeny)
        ols <- lm(y ~ x, data = mf)
        b_ols <- coef(ols)

        # Slope + p-value from the PGLS fit for the subtitle
        sm    <- summary(res$fit)$tTable
        slope <- signif(b_pgls[2], 3)
        pval  <- signif(sm[2, "p-value"], 3)

        ggplot(mf, aes(x = x, y = y)) +
          geom_point(size = 3, colour = "#8c6d3f") +
          geom_text(aes(label = species), vjust = -0.8, size = 3,
                    colour = "#5a4a2f") +
          geom_abline(aes(intercept = b_ols[1], slope = b_ols[2],
                          colour = "OLS (ignores phylogeny)"),
                      linewidth = 0.8, linetype = "dashed") +
          geom_abline(aes(intercept = b_pgls[1], slope = b_pgls[2],
                          colour = "PGLS (phylogeny-corrected)"),
                      linewidth = 1.0) +
          scale_colour_manual(name = NULL,
            values = c("OLS (ignores phylogeny)" = "#9e9e9e",
                       "PGLS (phylogeny-corrected)" = "#d7191c")) +
          labs(x = res$pred, y = res$resp,
               title = paste0("PGLS regression: ", res$resp, " ~ ", res$pred),
               subtitle = paste0("PGLS slope = ", slope, ",  p = ", pval,
                                 "   (dashed grey = uncorrected OLS)")) +
          theme_minimal(base_size = 13) +
          theme(legend.position = "top")

      } else {
        # Categorical predictor: boxplot + points, with group means marked
        grp_means <- aggregate(y ~ x, data = mf, FUN = mean)
        ggplot(mf, aes(x = x, y = y)) +
          geom_boxplot(fill = "#f5f0e8", colour = "#8c6d3f", outlier.shape = NA) +
          geom_jitter(aes(label = species), width = 0.15, size = 3,
                      colour = "#8c6d3f") +
          geom_point(data = grp_means, aes(x = x, y = y),
                     shape = 18, size = 5, colour = "#d7191c") +
          labs(x = res$pred, y = res$resp,
               title = paste0("PGLS: ", res$resp, " by ", res$pred),
               subtitle = "Red diamonds = group means; full PGLS model in summary tab") +
          theme_minimal(base_size = 13)
      }

    } else if (res$analysis == "bin_num") {
      # Binary response vs numeric predictor: points at 0/1 plus fitted
      # probability curves (naive logistic and phylogenetic logistic).
      gl <- suppressWarnings(glm(y ~ x, data = mf, family = binomial))
      xs <- seq(min(mf$x), max(mf$x), length.out = 200)
      curves <- data.frame(
        x = xs,
        p = as.numeric(plogis(coef(gl)[1] + coef(gl)[2] * xs)),
        model = "Logistic (ignores phylogeny)")
      subtitle <- "Dashed grey = logistic regression ignoring phylogeny"
      if (!failed) {
        b <- coef(res$fit)
        curves <- rbind(curves, data.frame(
          x = xs,
          p = as.numeric(plogis(b[1] + b[2] * xs)),
          model = "Phylogenetic logistic"))
        sub2 <- tryCatch({
          cf <- summary(res$fit)$coefficients
          paste0("Phylogenetic logistic slope = ",
                 signif(cf[2, "Estimate"], 3), ",  p = ",
                 signif(cf[2, "p.value"], 3),
                 "   (dashed grey = logistic ignoring phylogeny)")
        }, error = function(e) NULL)
        if (!is.null(sub2)) subtitle <- sub2
      } else {
        subtitle <- paste0(subtitle, "; phylogenetic fit failed (see PGLS Summary tab)")
      }
      ggplot(mf, aes(x = x, y = y)) +
        geom_point(position = position_jitter(width = 0, height = 0.03, seed = 1),
                   size = 3, colour = "#8c6d3f") +
        geom_line(data = curves, aes(x = x, y = p, colour = model,
                                     linetype = model), linewidth = 1) +
        scale_colour_manual(name = NULL,
          values = c("Logistic (ignores phylogeny)" = "#9e9e9e",
                     "Phylogenetic logistic"        = "#d7191c")) +
        scale_linetype_manual(name = NULL,
          values = c("Logistic (ignores phylogeny)" = "dashed",
                     "Phylogenetic logistic"        = "solid")) +
        scale_y_continuous(breaks = c(0, 0.5, 1),
          labels = c(paste0(res$nonevent, " (0)"), "0.5",
                     paste0(res$event, " (1)"))) +
        labs(x = res$pred, y = paste0("Probability of '", res$event, "'"),
             title = paste0("Logistic regression: ", res$resp, " ~ ", res$pred),
             subtitle = subtitle) +
        theme_minimal(base_size = 13) +
        theme(legend.position = "top")

    } else {
      # Binary response vs binary / categorical predictor: stacked
      # proportions per predictor group, labelled with species counts.
      ggplot(mf, aes(x = x, fill = y_lab)) +
        geom_bar(position = "fill", colour = "#8c6d3f") +
        geom_text(stat = "count", aes(label = after_stat(count)),
                  position = position_fill(vjust = 0.5), size = 5,
                  colour = "#3b2f1a") +
        scale_fill_manual(name = res$resp,
          values = setNames(c("#f5f0e8", "#d9b87c"),
                            c(res$nonevent, res$event))) +
        labs(x = res$pred, y = "Proportion of species",
             title = paste0(res$resp, " by ", res$pred),
             subtitle = "Numbers = species counts; tests are in the two summary tabs") +
        theme_minimal(base_size = 13) +
        theme(legend.position = "top")
    }
  })
}
