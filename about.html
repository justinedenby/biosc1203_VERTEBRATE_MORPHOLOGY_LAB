# ── Server ────────────────────────────────────────────────────────────────────
server <- function(input, output, session) {

  # Install packages in WebR on first run (runtime, browser-side).
  # DT and plotly are intentionally NOT used — their htmlwidgets runtime
  # throws a "data is null / deps" error under shinylive. Tables are rendered
  # as plain HTML and plots as static base/ggplot images instead.
  for (pkg in c("geomorph","ggplot2","readr","dplyr","stringr","ape","nlme")) {
    if (!requireNamespace(pkg, quietly = TRUE)) webr::install(pkg)
  }

  library(geomorph); library(ggplot2)
  library(readr);    library(dplyr); library(stringr)
  library(ape);      library(nlme)

  # ── helpers ─────────────────────────────────────────────────────────────────

  # Render a data frame as a simple scrollable HTML table (replaces DT).
  html_table <- function(df, digits = 4) {
    df <- as.data.frame(df)
    num <- vapply(df, is.numeric, logical(1))
    df[num] <- lapply(df[num], function(v) round(v, digits))
    hdr <- paste0("<th>", names(df), "</th>", collapse = "")
    rows <- apply(df, 1, function(r)
      paste0("<tr>", paste0("<td>", r, "</td>", collapse = ""), "</tr>"))
    HTML(paste0(
      "<div class='app-table'><table><thead><tr>", hdr,
      "</tr></thead><tbody>", paste0(rows, collapse = ""),
      "</tbody></table></div>"
    ))
  }

  read_csv_flexible <- function(path) {
    tryCatch({
      df <- read_csv(path, show_col_types = FALSE)
      if (ncol(df) < 3) df <- read_delim(path, delim = ";",
                                          show_col_types = FALSE)
      df
    }, error = function(e) NULL)
  }

  norm_species <- function(x) {
    x <- trimws(as.character(x))
    x <- gsub("\\s+", "_", x)
    tolower(x)
  }

  classify_landmarks <- function(df, label_col, x_col, y_col, z_col) {
    df |>
      select(label = all_of(label_col), x = all_of(x_col),
             y = all_of(y_col),         z = all_of(z_col)) |>
      mutate(
        label = as.character(label),
        label = if_else(is.na(label) | label == "",
                        paste0("unlabeled_", row_number()), label),
        x = as.numeric(x), y = as.numeric(y), z = as.numeric(z),
        type = case_when(
          str_detect(label, "^OC_")        ~ "semi",
          str_detect(label, "^unlabeled_") ~ "semi",
          TRUE                              ~ "fixed"
        )
      ) |>
      filter(!is.na(x), !is.na(y), !is.na(z))
  }

  build_tps <- function(lm_df, specimen_name) {
    fmt <- function(v) formatC(v, format = "f", digits = 4, width = 12,
                                flag = " ")
    coord_lines <- paste(fmt(lm_df$x), fmt(lm_df$y), fmt(lm_df$z), sep = "  ")
    c(paste0("LM3=", nrow(lm_df)), coord_lines, paste0("ID=", specimen_name))
  }

  parse_tps_file <- function(path) {
    lines <- readLines(path, warn = FALSE)
    lines <- lines[nzchar(trimws(lines))]
    specimens <- list(); spec_ids <- character(); i <- 1
    while (i <= length(lines)) {
      line <- trimws(lines[i])
      if (grepl("^LM3?=", line, ignore.case = TRUE)) {
        n_lm   <- as.integer(sub("^LM3?=", "", line, ignore.case = TRUE))
        coords <- matrix(NA_real_, nrow = n_lm, ncol = 3)
        for (j in seq_len(n_lm)) {
          i <- i + 1
          vals <- as.numeric(strsplit(trimws(lines[i]), "\\s+")[[1]])
          coords[j, ] <- vals[1:3]
        }
        spec_id <- paste0("specimen_", length(specimens) + 1)
        while (i < length(lines)) {
          i <- i + 1; next_line <- trimws(lines[i])
          if (grepl("^ID=", next_line, ignore.case = TRUE)) {
            spec_id <- sub("^ID=", "", next_line, ignore.case = TRUE); break
          }
          if (grepl("^LM3?=", next_line, ignore.case = TRUE)) {
            i <- i - 1; break
          }
        }
        specimens[[length(specimens) + 1]] <- coords
        spec_ids <- c(spec_ids, spec_id)
      } else { i <- i + 1 }
    }
    if (length(specimens) == 0) return(NULL)
    p <- nrow(specimens[[1]]); n <- length(specimens)
    A <- array(NA_real_, dim = c(p, 3, n))
    for (k in seq_len(n)) A[, , k] <- specimens[[k]]
    dimnames(A)[[3]] <- spec_ids
    list(A = A, ids = spec_ids, n_lm = p, n_spec = n)
  }

  master_tree <- reactive({
    tryCatch(read.tree("data/rodent_tree.nwk"), error = function(e) NULL)
  })

  # Cache the normalized tip labels once so every downstream output
  # (match summary, both tree plots, PGLS) reuses them instead of
  # recomputing norm_species() over the whole tree each render — this
  # matters under WebR where each call is slow.
  master_tips_norm <- reactive({
    tr <- master_tree(); req(tr)
    norm_species(tr$tip.label)
  })

  # ── section 1 & 2: CSV / parse / TPS ────────────────────────────────────────

  raw_data <- reactive({ req(input$csv_file); read_csv_flexible(input$csv_file$datapath) })

  output$col_pickers <- renderUI({
    df <- raw_data(); req(df); cols <- names(df)
    guess <- function(p) {
      m <- cols[str_detect(tolower(cols), paste(tolower(p), collapse = "|"))]
      if (length(m)) m[1] else cols[1]
    }
    tagList(
      selectInput("col_label", "Label column", choices = cols,
                  selected = guess(c("label","name","id","lm"))),
      selectInput("col_x", "X / R coord", choices = cols,
                  selected = guess(c("^r$","_x$","x$","r$"))),
      selectInput("col_y", "Y / A coord", choices = cols,
                  selected = guess(c("^a$","_y$","y$","a$"))),
      selectInput("col_z", "Z / S coord", choices = cols,
                  selected = guess(c("^s$","_z$","z$","s$")))
    )
  })

  output$tbl_raw <- renderUI({ req(raw_data()); html_table(head(raw_data(), 100)) })
  output$tbl_raw_ready <- reactive({
    isTRUE(tryCatch(!is.null(raw_data()), error = function(e) FALSE))
  })
  outputOptions(output, "tbl_raw_ready", suspendWhenHidden = FALSE)

  parsed <- eventReactive(input$run_parse, {
    req(raw_data(), input$col_label, input$col_x, input$col_y, input$col_z)
    classify_landmarks(raw_data(), input$col_label,
                       input$col_x, input$col_y, input$col_z)
  })

  output$tbl_parsed <- renderUI({ req(parsed()); html_table(parsed()) })

  output$lm_summary <- renderPrint({
    req(parsed()); df <- parsed()
    cat("Total landmarks:", nrow(df), "\n")
    cat("Fixed landmarks:", sum(df$type == "fixed"), "\n")
    cat("Semilandmarks  :", sum(df$type == "semi"),  "\n\n")
    cat("Fixed landmark labels:\n")
    cat(paste(" ", df$label[df$type == "fixed"], collapse = "\n"), "\n\n")
    cat("Semilandmark labels:\n")
    cat(paste(" ", df$label[df$type == "semi"],  collapse = "\n"), "\n")
  })

  tps_lines <- reactive({ req(parsed()); build_tps(parsed(), input$specimen_name) })
  output$tps_preview <- renderText({ req(tps_lines()); paste(tps_lines(), collapse = "\n") })
  output$dl_tps <- downloadHandler(
    filename = function() paste0(input$specimen_name, ".tps"),
    content  = function(file) writeLines(tps_lines(), file)
  )

  # ── section 3: TPS analysis ─────────────────────────────────────────────────

  tps_parsed <- eventReactive(input$tps_file, {
    req(input$tps_file)
    result <- tryCatch(parse_tps_file(input$tps_file$datapath),
                       error = function(e) NULL)
    validate(need(!is.null(result),
      "Could not parse TPS file. Check format: LM3=n / coords / ID=name."))
    result
  })

  output$tbl_tps_specimens <- renderUI({
    req(tps_parsed()); tp <- tps_parsed()
    html_table(data.frame(Index = seq_along(tp$ids), Specimen = tp$ids,
                          N_landmarks = tp$n_lm))
  })
  output$tbl_tps_ready <- reactive({
    isTRUE(tryCatch(!is.null(input$tps_file), error = function(e) FALSE))
  })
  outputOptions(output, "tbl_tps_ready", suspendWhenHidden = FALSE)

  tps_results <- eventReactive(input$run_tps_analysis, {
    req(tps_parsed()); tp <- tps_parsed()
    validate(need(tp$n_spec >= 2,
      paste0("GPA requires at least 2 specimens. Your file has ", tp$n_spec, ".")))
    gpa <- tryCatch(
      gpagen(tp$A, curves = NULL, ProcD = (input$tps_procD == "full"),
             print.progress = FALSE),
      error = function(e) validate(need(FALSE, paste("GPA failed:", e$message))))
    pca <- tryCatch(gm.prcomp(gpa$coords),
      error = function(e) validate(need(FALSE, paste("PCA failed:", e$message))))
    list(gpa = gpa, pca = pca, ids = tp$ids)
  })

  output$tps_gpa_summary <- renderPrint({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); res <- tps_results()
    cat("=== GPA (Generalized Procrustes Analysis) ===\n\n")
    print(summary(res$gpa))
    cat("\nConsensus shape (first 6 landmarks):\n")
    print(round(head(res$gpa$consensus, 6), 4))
  })

  output$tbl_tps_pca <- renderUI({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); pca <- tps_results()$pca
    eigs <- pca$sdev^2; var_exp <- round(eigs / sum(eigs) * 100, 2)
    n_pcs <- min(input$tps_n_pcs, length(var_exp))
    scores_df <- as.data.frame(pca$x[, 1:n_pcs, drop = FALSE])
    names(scores_df) <- paste0("PC", 1:n_pcs, " (", var_exp[1:n_pcs], "%)")
    scores_df <- cbind(Specimen = rownames(scores_df), scores_df)
    html_table(scores_df)
  })

  output$plot_tps_pca <- renderPlot({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); pca <- tps_results()$pca
    eigs <- pca$sdev^2; var_exp <- round(eigs / sum(eigs) * 100, 2)
    scores <- as.data.frame(pca$x); scores$Specimen <- rownames(scores)
    ggplot(scores, aes(x = Comp1, y = Comp2, label = Specimen)) +
      geom_point(size = 3, colour = "#2c7bb6") +
      geom_text(vjust = -0.8, size = 3) +
      labs(x = paste0("PC1 (", var_exp[1], "% variance)"),
           y = paste0("PC2 (", var_exp[2], "% variance)"),
           title = "PCA morphospace — PC1 vs PC2") +
      theme_minimal(base_size = 13)
  })

  output$plot_tps_scree <- renderPlot({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); pca <- tps_results()$pca
    eigs <- pca$sdev^2; var_exp <- round(eigs / sum(eigs) * 100, 2)
    n_show <- min(input$tps_n_pcs, length(var_exp))
    df_sc <- data.frame(
      PC = factor(paste0("PC", 1:n_show), levels = paste0("PC", 1:n_show)),
      Variance = var_exp[1:n_show])
    ggplot(df_sc, aes(x = PC, y = Variance)) +
      geom_col(fill = "#2c7bb6", alpha = 0.8) +
      geom_line(aes(group = 1), colour = "#d7191c", linewidth = 0.8) +
      geom_point(colour = "#d7191c", size = 2) +
      labs(title = "Scree plot", y = "% Variance", x = "") +
      theme_minimal(base_size = 13)
  })

  output$plot_tps_consensus <- renderPlot({
    validate(need(input$run_tps_analysis > 0, "Click 'Run GPA + PCA' to see results."))
    req(tps_results()); cons <- tps_results()$gpa$consensus
    df_plot <- data.frame(x = cons[,1], y = cons[,2], z = cons[,3],
                          label = rownames(cons))
    par(mfrow = c(1, 2), mar = c(4, 4, 3, 1))
    plot(df_plot$x, df_plot$y, pch = 19, col = "#2c7bb6", cex = 1.2,
         xlab = "X (R)", ylab = "Y (A)", main = "Consensus — R-A view")
    text(df_plot$x, df_plot$y, labels = df_plot$label, cex = 0.6, pos = 3)
    plot(df_plot$x, df_plot$z, pch = 19, col = "#2c7bb6", cex = 1.2,
         xlab = "X (R)", ylab = "Z (S)", main = "Consensus — R-S view")
    text(df_plot$x, df_plot$z, labels = df_plot$label, cex = 0.6, pos = 3)
  })

  # ── section 4: phylogenetic analyses (PGLS) ─────────────────────────────────

  trait_data <- reactive({
    req(input$trait_file)
    df <- read_csv_flexible(input$trait_file$datapath)
    validate(need(!is.null(df) && ncol(df) >= 2,
      "Trait CSV needs a species column plus at least one trait column."))
    names(df)[1] <- "species_raw"
    df$species_match <- norm_species(df$species_raw)
    df
  })

  output$trait_ready <- reactive({
    isTRUE(tryCatch(!is.null(input$trait_file), error = function(e) FALSE))
  })
  outputOptions(output, "trait_ready", suspendWhenHidden = FALSE)

  trait_cols <- reactive({
    df <- trait_data()
    setdiff(names(df), c("species_raw", "species_match"))
  })

  trait_types <- reactive({
    df <- trait_data(); cols <- trait_cols()
    sapply(cols, function(cc) if (is.numeric(df[[cc]])) "numeric" else "categorical")
  })

  output$tbl_traits <- renderUI({
    req(trait_data())
    html_table(trait_data() |> select(-species_match))
  })

  output$pgls_response_ui <- renderUI({
    req(trait_data())
    tt <- trait_types(); num_cols <- names(tt)[tt == "numeric"]
    validate(need(length(num_cols) >= 1,
      "Need at least one numeric trait to use as the response."))
    selectInput("pgls_response", "Response (Y, continuous)",
                choices = num_cols, selected = num_cols[1])
  })

  output$pgls_predictor_ui <- renderUI({
    req(trait_data(), input$pgls_response)
    tt      <- trait_types()
    choices <- setdiff(trait_cols(), input$pgls_response)
    validate(need(length(choices) >= 1, "Need at least one other trait as predictor."))

    num_choices <- choices[tt[choices] == "numeric"]
    cat_choices <- choices[tt[choices] == "categorical"]

    # Group the dropdown so students see the type of each predictor.
    # Numeric predictors are listed first and selected by default so the
    # regression line shows unless they deliberately pick a category.
    grouped <- list()
    if (length(num_choices))
      grouped[["Numeric → regression line"]] <- as.list(num_choices)
    if (length(cat_choices))
      grouped[["Categorical → group comparison"]] <- as.list(cat_choices)

    default <- if (length(num_choices)) num_choices[1] else cat_choices[1]

    tagList(
      selectInput("pgls_predictor", "Predictor (X)",
                  choices = grouped, selected = default),
      uiOutput("pgls_plot_hint")
    )
  })

  # Live hint under the predictor dropdown telling the student what they'll get.
  output$pgls_plot_hint <- renderUI({
    req(input$pgls_predictor)
    tt <- trait_types()
    if (isTRUE(tt[[input$pgls_predictor]] == "numeric")) {
      div(style = "font-size:0.8rem; color:#5a4a2f; margin-top:-8px;",
          "Numeric predictor → the PGLS plot shows a fitted regression line.")
    } else {
      div(style = "font-size:0.8rem; color:#5a4a2f; margin-top:-8px;",
          "Categorical predictor → no regression line (a line through categories
           isn't meaningful). The plot shows group means instead; the PGLS
           model is still fit and reported in the summary tab.")
    }
  })

  tree_match <- reactive({
    req(trait_data(), master_tree())
    tr <- master_tree()
    validate(need(!is.null(tr), "Master tree could not be loaded from data/rodent_tree.nwk."))
    tips_norm  <- master_tips_norm()
    trait_norm <- trait_data()$species_match
    matched    <- intersect(trait_norm, tips_norm)
    list(tree = tr, tips_norm = tips_norm, trait_norm = trait_norm,
         matched = matched,
         unmatched = setdiff(trait_norm, tips_norm),
         in_tree_only = setdiff(tips_norm, trait_norm))
  })

  output$tree_match_summary <- renderPrint({
    req(tree_match()); tm <- tree_match()
    cat("Master tree tips:", length(tm$tips_norm), "\n")
    cat("Species in trait CSV:", length(tm$trait_norm), "\n")
    cat("Matched to tree:", length(tm$matched), "\n\n")
    if (length(tm$unmatched) > 0) {
      cat("In CSV but NOT in tree (check spelling):\n")
      cat(paste(" ", tm$unmatched, collapse = "\n"), "\n\n")
    } else cat("All CSV species matched the tree.\n\n")
    cat("Matched species used in analyses:\n")
    cat(paste(" ", tm$matched, collapse = "\n"), "\n")
  })

  output$plot_master_tree <- renderPlot({
    req(tree_match()); tm <- tree_match(); tr <- tm$tree
    is_match <- master_tips_norm() %in% tm$matched
    tip_col  <- ifelse(is_match, "#8c6d3f", "#b8b8b8")
    tip_font <- ifelse(is_match, 2, 1)
    par(mar = c(1, 1, 3, 1))   # leave room at top for the title
    plot(tr, tip.color = tip_col, font = tip_font, cex = 0.9)
    title(main = "Master rodent tree — uploaded species in brown/bold",
          cex.main = 1.1, line = 1)
  })

  pruned_tree <- reactive({
    req(tree_match()); tm <- tree_match(); tr <- tm$tree
    validate(need(length(tm$matched) >= 3,
      "Need at least 3 matched species to prune and analyse."))
    keep_idx <- which(master_tips_norm() %in% tm$matched)
    drop_idx <- setdiff(seq_along(tr$tip.label), keep_idx)
    if (length(drop_idx) == 0) tr else drop.tip(tr, tr$tip.label[drop_idx])
  })

  output$plot_pruned_tree <- renderPlot({
    req(pruned_tree())
    par(mar = c(1, 1, 3, 1))   # leave room at top for the title
    plot(pruned_tree(), tip.color = "#8c6d3f", font = 2, cex = 1.0)
    title(main = "Pruned tree — matched species only",
          cex.main = 1.1, line = 1)
  })

  pgls_fit <- eventReactive(input$run_pgls, {
    req(trait_data(), pruned_tree(), input$pgls_response, input$pgls_predictor)
    tr <- pruned_tree(); df <- trait_data()
    df <- df[df$species_match %in% norm_species(tr$tip.label), , drop = FALSE]
    df <- df[!duplicated(df$species_match), , drop = FALSE]
    rownames(df) <- df$species_match
    tr$tip.label <- norm_species(tr$tip.label)

    resp <- input$pgls_response; pred <- input$pgls_predictor
    mf <- data.frame(
      species = rownames(df),
      y = as.numeric(df[[resp]]),
      x = if (trait_types()[[pred]] == "numeric") as.numeric(df[[pred]])
          else factor(df[[pred]]),
      stringsAsFactors = FALSE
    )
    rownames(mf) <- mf$species
    mf <- mf[complete.cases(mf[, c("y", "x")]), , drop = FALSE]

    # Hard floor: PGLS needs to estimate an intercept, a slope, and a
    # residual variance, so it cannot run below 4 species and is not
    # trustworthy much below that. Block clearly rather than fit garbage.
    validate(need(nrow(mf) >= 4,
      paste0("Only ", nrow(mf), " species have complete data for this pair. ",
             "PGLS needs at least 4 (and ideally 10+) to give a meaningful fit. ",
             "Add more species or choose traits with fewer missing values.")))

    # For a categorical predictor, every group needs at least 2 species,
    # otherwise gls cannot estimate that group and the fit fails or misleads.
    if (trait_types()[[pred]] == "categorical") {
      tab <- table(mf$x)
      validate(need(all(tab >= 2) && length(tab) >= 2,
        paste0("Each category needs at least 2 species (and there must be at ",
               "least 2 categories). Current counts: ",
               paste(names(tab), as.integer(tab), sep = "=", collapse = ", "),
               ". Pick a different predictor or add species.")))
    }

    tr2 <- drop.tip(tr, setdiff(tr$tip.label, rownames(mf)))
    mf  <- mf[tr2$tip.label, , drop = FALSE]

    cor_struct <- if (input$pgls_model == "lambda") {
      corPagel(value = 0.5, phy = tr2, form = ~species, fixed = FALSE)
    } else {
      corBrownian(phy = tr2, form = ~species)
    }

    # Fit PGLS. If corPagel fails to optimise lambda (common on small or
    # low-signal data), retry with lambda fixed at a middling value so the
    # student still gets a result instead of a blank screen. Capture any
    # error message so it can be shown rather than silently swallowed.
    fit <- tryCatch(
      gls(y ~ x, data = mf, correlation = cor_struct, method = "ML"),
      error = function(e) {
        tryCatch(
          gls(y ~ x, data = mf,
              correlation = corPagel(0.5, phy = tr2, form = ~species,
                                     fixed = TRUE),
              method = "ML"),
          error = function(e2)
            structure(list(msg = conditionMessage(e)), class = "pgls_error")
        )
      })

    list(fit = fit, mf = mf, resp = resp, pred = pred,
         pred_type = trait_types()[[pred]])
  })

  output$pgls_summary <- renderPrint({
    validate(need(input$run_pgls > 0, "Choose variables and click 'Run PGLS'."))
    req(pgls_fit()); res <- pgls_fit()
    if (inherits(res$fit, "pgls_error")) {
      cat("PGLS could not be fit for this combination.\n\n")
      cat("Error reported by gls():\n  ", res$fit$msg, "\n\n")
      cat("Tips: try the 'Brownian motion' correlation instead of Pagel's\n")
      cat("lambda, or choose a different response/predictor pair.\n")
      return(invisible())
    }
    cat("=== Phylogenetic Generalized Least Squares (nlme::gls) ===\n")
    cat("Response   :", res$resp, "\n")
    cat("Predictor  :", res$pred, " (", res$pred_type, ")\n")
    cat("Correlation:", input$pgls_model, "\n")
    cat("N species  :", nrow(res$mf), "\n\n")
    if (nrow(res$mf) < 10) {
      cat("NOTE: With fewer than 10 species, PGLS estimates (especially\n")
      cat("Pagel's lambda and p-values) are unstable. Interpret with caution.\n\n")
    }
    print(summary(res$fit))
  })

  output$plot_pgls <- renderPlot({
    validate(need(input$run_pgls > 0, "Choose variables and click 'Run PGLS'."))
    req(pgls_fit()); res <- pgls_fit(); mf <- res$mf
    validate(need(!inherits(res$fit, "pgls_error"),
      paste0("PGLS could not be fit: ", if (inherits(res$fit, "pgls_error"))
             res$fit$msg else "", "\nSee the PGLS summary tab for details.")))
    mf$species <- rownames(mf)

    if (res$pred_type == "numeric") {
      # Guard: a numeric predictor needs enough distinct values for a
      # meaningful regression line. If not, tell the student rather than
      # drawing a misleading fit.
      validate(need(length(unique(mf$x)) >= 3,
        paste0("'", res$pred, "' has fewer than 3 distinct numeric values, ",
               "so a regression line would be misleading. Pick a predictor ",
               "with more variation, or treat this variable as categorical.")))

      # PGLS (phylogeny-corrected) coefficients
      b_pgls <- coef(res$fit)
      # Ordinary least squares for comparison (ignores phylogeny)
      ols <- lm(y ~ x, data = mf)
      b_ols <- coef(ols)

      # Slope + p-value from the PGLS fit for the subtitle
      sm    <- summary(res$fit)$tTable
      slope <- signif(b_pgls[2], 3)
      pval  <- signif(sm[2, "p-value"], 3)

      ggplot(mf, aes(x = x, y = y)) +
        geom_point(size = 3, colour = "#8c6d3f") +
        geom_text(aes(label = species), vjust = -0.8, size = 3,
                  colour = "#5a4a2f") +
        geom_abline(aes(intercept = b_ols[1], slope = b_ols[2],
                        colour = "OLS (ignores phylogeny)"),
                    linewidth = 0.8, linetype = "dashed") +
        geom_abline(aes(intercept = b_pgls[1], slope = b_pgls[2],
                        colour = "PGLS (phylogeny-corrected)"),
                    linewidth = 1.0) +
        scale_colour_manual(name = NULL,
          values = c("OLS (ignores phylogeny)" = "#9e9e9e",
                     "PGLS (phylogeny-corrected)" = "#d7191c")) +
        labs(x = res$pred, y = res$resp,
             title = paste0("PGLS regression: ", res$resp, " ~ ", res$pred),
             subtitle = paste0("PGLS slope = ", slope, ",  p = ", pval,
                               "   (dashed grey = uncorrected OLS)")) +
        theme_minimal(base_size = 13) +
        theme(legend.position = "top")

    } else {
      # Categorical predictor: boxplot + points, with group means marked
      grp_means <- aggregate(y ~ x, data = mf, FUN = mean)
      ggplot(mf, aes(x = x, y = y)) +
        geom_boxplot(fill = "#f5f0e8", colour = "#8c6d3f", outlier.shape = NA) +
        geom_jitter(aes(label = species), width = 0.15, size = 3,
                    colour = "#8c6d3f") +
        geom_point(data = grp_means, aes(x = x, y = y),
                   shape = 18, size = 5, colour = "#d7191c") +
        labs(x = res$pred, y = res$resp,
             title = paste0("PGLS: ", res$resp, " by ", res$pred),
             subtitle = "Red diamonds = group means; full PGLS model in summary tab") +
        theme_minimal(base_size = 13)
    }
  })
}
